# This script DELETES all generated ingestion artifacts and runtime DB content.
# It preserves data\raw, model caches, database schema, and schema_migrations.
# Run the complete file from top to bottom. Do not run ingestion in parallel.
# Execute this TXT file with:
# Invoke-Expression ([System.IO.File]::ReadAllText((Resolve-Path "FULL_INGESTION_COMMANDS.txt"), [System.Text.Encoding]::UTF8))
# To preserve completed work and resume from one source type:
# $env:INGESTION_RESUME_FROM="standard"
# Invoke-Expression ([System.IO.File]::ReadAllText((Resolve-Path "FULL_INGESTION_COMMANDS.txt"), [System.Text.Encoding]::UTF8))

$PersistScript = Join-Path $Root "scripts\persist_ingestion_exports.py"
$ResetScript = Join-Path $Root "scripts\reset_runtime_stores.py"

function Assert-ExitCode {
    param(
        [Parameter(Mandatory = $true)][int]$Code,
        [Parameter(Mandatory = $true)][string]$Step,
        [Parameter(Mandatory = $true)][string]$Log
    )
    if ($Code -ne 0) {
        throw "$Step failed with exit code $Code. See $Log"
    }
}

function Invoke-ExternalLogged {
    param(
        [Parameter(Mandatory = $true)][string]$Executable,
        [Parameter(Mandatory = $true)][object[]]$ArgumentList,
        [Parameter(Mandatory = $true)][string]$Log
    )
    $previousErrorAction = $ErrorActionPreference
    try {
        # Windows PowerShell 5.1 wraps native stderr as NativeCommandError.
        # Record stderr and trust the process exit code instead.
        $ErrorActionPreference = "Continue"
        & $Executable @ArgumentList 2>&1 | Out-File -FilePath $Log -Encoding utf8
        return $LASTEXITCODE
    } finally {
        $ErrorActionPreference = $previousErrorAction
    }
}

function Read-Utf8Json {
    param([Parameter(Mandatory = $true)][string]$Path)
    $full = (Resolve-Path -LiteralPath $Path).Path
    return ([System.IO.File]::ReadAllText($full, [System.Text.Encoding]::UTF8) | ConvertFrom-Json)
}

function Assert-JsonOk {
    param(
        [Parameter(Mandatory = $true)][string]$Path,
        [Parameter(Mandatory = $true)][string]$Step
    )
    $payload = Read-Utf8Json -Path $Path
    if ($payload.status -ne "ok") {
        throw "$Step returned status '$($payload.status)'. See $Path"
    }
    return $payload
}

function Clear-GeneratedDirectory {
    param([Parameter(Mandatory = $true)][string]$RelativePath)
    $full = [System.IO.Path]::GetFullPath((Join-Path $Root $RelativePath))
    $prefix = $DataRoot + [System.IO.Path]::DirectorySeparatorChar
    if (-not $full.StartsWith($prefix, [System.StringComparison]::OrdinalIgnoreCase)) {
        throw "Unsafe cleanup target: $full"
    }
    if (-not (Test-Path -LiteralPath $full)) {
        New-Item -ItemType Directory -Path $full -Force | Out-Null
        return
    }
    $children = @(Get-ChildItem -LiteralPath $full -Force | Where-Object { $_.Name -ne ".gitkeep" })
    foreach ($child in $children) {
        if (-not $child.FullName.StartsWith($full + [System.IO.Path]::DirectorySeparatorChar, [System.StringComparison]::OrdinalIgnoreCase)) {
            throw "Unsafe cleanup child: $($child.FullName)"
        }
        Remove-Item -LiteralPath $child.FullName -Recurse -Force
    }
}

function Invoke-NativeDocument {
    param(
        [Parameter(Mandatory = $true)][string]$SourceType,
        [Parameter(Mandatory = $true)][object]$Document,
        [Parameter(Mandatory = $true)][int]$Index
    )
    $sourcePath = [string]$Document.source_path
    if (-not (Test-Path -LiteralPath $sourcePath)) {
        throw "Native source path does not exist: $sourcePath"
    }
    $tag = "{0}_native_{1:D3}" -f $SourceType, $Index
    $output = Join-Path $LogDir ($tag + ".json")
    $diagnostic = Join-Path $LogDir ($tag + ".diagnostic.txt")
    Write-Host "[$SourceType] native $Index`: $([System.IO.Path]::GetFileName($sourcePath))"
    $arguments = @("-m", "apps.cli.main", "ingest", $sourcePath, "--source-type", $SourceType, "--output", $output)
    $code = Invoke-ExternalLogged -Executable $Python -ArgumentList $arguments -Log $diagnostic
    Assert-ExitCode -Code $code -Step "$SourceType native item $Index" -Log $diagnostic
    $null = Assert-JsonOk -Path $output -Step "$SourceType native item $Index"
}

function Invoke-OcrCheckpoint {
    param(
        [Parameter(Mandatory = $true)][string]$SourceType,
        [Parameter(Mandatory = $true)][string]$StagingDirectory,
        [Parameter(Mandatory = $true)][string]$Tag,
        [object]$MaxPages = $null
    )
    $suffix = if ($null -eq $MaxPages) { "full" } else { "{0:D4}" -f [int]$MaxPages }
    $log = Join-Path $LogDir ($Tag + "_" + $suffix + ".txt")
    $ocrArgs = @(
        $OcrScript,
        "--raw-dir", $StagingDirectory,
        "--source-type", $SourceType,
        "--doc-id-mode", "content"
    )
    if ($null -ne $MaxPages) {
        $ocrArgs += @("--max-pages", [string][int]$MaxPages)
    }
    $code = Invoke-ExternalLogged -Executable $Python -ArgumentList $ocrArgs -Log $log
    Assert-ExitCode -Code $code -Step "$Tag checkpoint $suffix" -Log $log
}

function Invoke-OcrDocument {
    param(
        [Parameter(Mandatory = $true)][string]$SourceType,
        [Parameter(Mandatory = $true)][object]$Document,
        [Parameter(Mandatory = $true)][int]$Index
    )
    $sourcePath = [string]$Document.source_path
    if (-not (Test-Path -LiteralPath $sourcePath)) {
        throw "OCR source path does not exist: $sourcePath"
    }

    $staging = Join-Path $StagingRoot $SourceType
    Clear-GeneratedDirectory -RelativePath ("data\tmp\ingestion_staging\" + $SourceType)
    Copy-Item -LiteralPath $sourcePath -Destination $staging -Force

    $pageCount = 0
    if ($null -ne $Document.page_count) {
        $pageCount = [int]$Document.page_count
    }
    $tag = "{0}_ocr_{1:D3}" -f $SourceType, $Index
    Write-Host "[$SourceType] OCR $Index ($pageCount pages): $([System.IO.Path]::GetFileName($sourcePath))"

    foreach ($checkpoint in @(25, 50, 100, 200, 400, 800)) {
        if ($pageCount -gt $checkpoint) {
            Invoke-OcrCheckpoint -SourceType $SourceType -StagingDirectory $staging -Tag $tag -MaxPages $checkpoint
        }
    }
    Invoke-OcrCheckpoint -SourceType $SourceType -StagingDirectory $staging -Tag $tag -MaxPages $null

    $docId = [string]$Document.doc_id
    $manifest = Join-Path $DataRoot ("exports\" + $SourceType + "\" + $docId + "\manifest.json")
    if (-not (Test-Path -LiteralPath $manifest)) {
        throw "OCR completed but manifest is missing: $manifest"
    }
    $manifestPayload = Read-Utf8Json -Path $manifest
    if ($manifestPayload.status -ne "ok") {
        throw "OCR manifest status is '$($manifestPayload.status)': $manifest"
    }
}

function Write-StoreCounts {
    param([Parameter(Mandatory = $true)][string]$Tag)
    $pgLog = Join-Path $LogDir ($Tag + "_postgres.txt")
    $qdrantLog = Join-Path $LogDir ($Tag + "_qdrant.json")
    $pgCountLog = Join-Path $LogDir ($Tag + "_postgres_embeddable_count.txt")

    $sql = "select count(*) as documents from documents; select count(*) as chunks from chunks; select count(*) as embeddable_chunks from chunks where btrim(coalesce(normalized_text,text,'')) <> ''; select type,count(*) from documents group by type order by type;"
    $arguments = @("compose", "exec", "-T", "postgres", "psql", "-U", "vib", "-d", "vibration", "-c", $sql)
    $code = Invoke-ExternalLogged -Executable "docker" -ArgumentList $arguments -Log $pgLog
    Assert-ExitCode -Code $code -Step "$Tag Postgres validation" -Log $pgLog

    $qdrant = Invoke-RestMethod -Method Post -Uri "http://localhost:6333/collections/chunks/points/count" -ContentType "application/json" -Body '{"exact":true}'
    $qdrant | ConvertTo-Json -Depth 10 | Out-File -FilePath $qdrantLog -Encoding utf8

    $countSql = "select count(*) from chunks where btrim(coalesce(normalized_text,text,'')) <> '';"
    $countArguments = @("compose", "exec", "-T", "postgres", "psql", "-U", "vib", "-d", "vibration", "-At", "-c", $countSql)
    $code = Invoke-ExternalLogged -Executable "docker" -ArgumentList $countArguments -Log $pgCountLog
    Assert-ExitCode -Code $code -Step "$Tag embeddable chunk count" -Log $pgCountLog
    $pgCountLines = @([System.IO.File]::ReadAllLines($pgCountLog, [System.Text.Encoding]::UTF8) | Where-Object { $_.Trim() })
    if ($pgCountLines.Count -eq 0) { throw "$Tag embeddable chunk count returned no output. See $pgCountLog" }
    $pgCount = [int64]$pgCountLines[-1].Trim()
    $qdrantCount = [int64]$qdrant.result.count
    if ($pgCount -ne $qdrantCount) {
        throw "$Tag storage mismatch: Postgres embeddable_chunks=$pgCount, Qdrant points=$qdrantCount"
    }
}

function Invoke-SourceType {
    param([Parameter(Mandatory = $true)][string]$SourceType)
    $rawDirectory = Join-Path $DataRoot ("raw\" + $SourceType)
    if (-not (Test-Path -LiteralPath $rawDirectory)) {
        throw "Raw directory does not exist: $rawDirectory"
    }

    $plan = Join-Path $LogDir ($SourceType + "_plan.json")
    $diagnostic = Join-Path $LogDir ($SourceType + "_plan.diagnostic.txt")
    Write-Host "[$SourceType] planning"
    $arguments = @("-m", "apps.cli.main", "ingest", $rawDirectory, "--plan-only", "--output", $plan)
    $code = Invoke-ExternalLogged -Executable $Python -ArgumentList $arguments -Log $diagnostic
    Assert-ExitCode -Code $code -Step "$SourceType plan" -Log $diagnostic
    $planPayload = Assert-JsonOk -Path $plan -Step "$SourceType plan"

    $documents = @($planPayload.documents | Sort-Object source_path)
    if ($documents.Count -eq 0) {
        throw "$SourceType plan contains no documents."
    }
    $missing = @($documents | Where-Object { -not (Test-Path -LiteralPath ([string]$_.source_path)) })
    if ($missing.Count -gt 0) {
        $missing | Select-Object source_path, processing_strategy |
            ConvertTo-Json -Depth 5 |
            Out-File -FilePath (Join-Path $LogDir ($SourceType + "_missing_paths.json")) -Encoding utf8
        throw "$SourceType plan contains missing source paths."
    }

    $nativeDocuments = @($documents | Where-Object { $_.processing_strategy -ne "ocr_pdf" })
    $ocrDocuments = @($documents | Where-Object { $_.processing_strategy -eq "ocr_pdf" })
    [pscustomobject]@{
        source_type = $SourceType
        total = $documents.Count
        native = $nativeDocuments.Count
        ocr = $ocrDocuments.Count
        pages = [int64](($documents | Measure-Object -Property page_count -Sum).Sum)
    } | ConvertTo-Json | Out-File -FilePath (Join-Path $LogDir ($SourceType + "_plan_summary.json")) -Encoding utf8

    $index = 0
    foreach ($document in $nativeDocuments) {
        $index += 1
        Invoke-NativeDocument -SourceType $SourceType -Document $document -Index $index
    }

    $index = 0
    foreach ($document in $ocrDocuments) {
        $index += 1
        Invoke-OcrDocument -SourceType $SourceType -Document $document -Index $index
    }

    $persistOutput = Join-Path $LogDir ($SourceType + "_persist.json")
    $persistDiagnostic = Join-Path $LogDir ($SourceType + "_persist.diagnostic.txt")
    $arguments = @($PersistScript, "--source-type", $SourceType, "--output", $persistOutput)
    $code = Invoke-ExternalLogged -Executable $Python -ArgumentList $arguments -Log $persistDiagnostic
    Assert-ExitCode -Code $code -Step "$SourceType export persistence" -Log $persistDiagnostic
    $null = Assert-JsonOk -Path $persistOutput -Step "$SourceType export persistence"

    Write-StoreCounts -Tag ($SourceType + "_after")
    Write-Host "[$SourceType] complete"
}

    $code = Invoke-ExternalLogged -Executable "docker" -ArgumentList @("compose", "exec", "-T", "postgres", "pg_isready", "-U", "vib", "-d", "vibration") -Log $postgresReadyLog

            $qdrantReady = $true

        }
